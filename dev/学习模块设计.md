```sql
-- ======================================================================
-- 1. 课程/技能树主表 (Courses)
-- 对应 UI 中的 "Shell", "React", "Linux" 等卡片
-- ======================================================================
CREATE TABLE `courses` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `title` varchar(128) NOT NULL DEFAULT '' COMMENT '课程标题 (如 Shell)',
  `slug` varchar(128) NOT NULL DEFAULT '' COMMENT 'URL Slug (唯一, 如 shell-programming)',
  `description` varchar(512) NOT NULL DEFAULT '' COMMENT '简短介绍',
  `icon_url` varchar(512) NOT NULL DEFAULT '' COMMENT '图标/Logo URL',
  `cover_color` varchar(32) NOT NULL DEFAULT '#3b82f6' COMMENT '主题色 (UI卡片背景色)',
  
  -- 统计/展示数据 (UI显示用)
  `difficulty` tinyint unsigned NOT NULL DEFAULT '1' COMMENT '难度: 1-初级, 2-中级, 3-高级',
  `total_lessons` int unsigned NOT NULL DEFAULT '0' COMMENT '包含的总课程单元数',
  `total_skills` int unsigned NOT NULL DEFAULT '0' COMMENT '包含的技能点数 (如 26 Skills)',
  `estimated_hours` int unsigned NOT NULL DEFAULT '0' COMMENT '预计学习时长(小时)',
  
  -- 状态控制
  `is_pro` tinyint unsigned NOT NULL DEFAULT '0' COMMENT '是否为 Pro 专属课程',
  `status` tinyint unsigned NOT NULL DEFAULT '1' COMMENT '状态: 0-草稿, 1-发布, 2-下架',
  
  `created_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  `updated_at` datetime(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6) ON UPDATE CURRENT_TIMESTAMP(6),
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_course_slug` (`slug`),
  KEY `idx_status_sort` (`status`, `is_pro`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='课程/技能树主表';

-- ======================================================================
-- 2. 课程阶段表 (Course Stages)
-- 对应 UI 中的 "Shell Beginner", "Shell Intermediate" 等分层
-- ======================================================================
CREATE TABLE `course_stages` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `course_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '关联 courses.id',
  
  `title` varchar(128) NOT NULL DEFAULT '' COMMENT '阶段名称',
  `description` varchar(512) NOT NULL DEFAULT '' COMMENT '阶段描述 (掌握核心概念...)',
  `sort_order` int unsigned NOT NULL DEFAULT '0' COMMENT '排序权重 (从小到大)',
  `level_label` varchar(32) NOT NULL DEFAULT '' COMMENT '难度标签 (如 BEGINNER)',
  
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  
  PRIMARY KEY (`id`),
  KEY `idx_course_order` (`course_id`, `sort_order`) COMMENT '查询课程详情时按此排序'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='课程阶段划分';

-- ======================================================================
-- 3. 课程单元/课时表 (Course Lessons)
-- 对应 UI 中的 "Lab 1: First Shell Script", "Challenge: Log Analysis"
-- ======================================================================
CREATE TABLE `course_lessons` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `course_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '冗余字段，方便查询',
  `stage_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '关联 course_stages.id',
  
  -- 内容定义
  `title` varchar(128) NOT NULL DEFAULT '' COMMENT '课时标题',
  `type` tinyint unsigned NOT NULL DEFAULT '1' COMMENT '类型: 1-实验(Lab), 2-挑战(Challenge), 3-测验(Quiz), 4-文本(Text)',
  `duration_minutes` int unsigned NOT NULL DEFAULT '15' COMMENT '预计耗时(分钟)',
  
  -- 关联实际内容的标识
  -- 如果是实验/挑战，这里填 Lab 环境的 ID
  `lab_identifier` varchar(64) NOT NULL DEFAULT '' COMMENT '关联的实验环境/题目ID',
  
  `points` int unsigned NOT NULL DEFAULT '10' COMMENT '完成获得的经验值/技能点',
  `sort_order` int unsigned NOT NULL DEFAULT '0' COMMENT '组内排序',
  
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  
  PRIMARY KEY (`id`),
  KEY `idx_stage_content` (`stage_id`, `sort_order`),
  KEY `idx_course_lookup` (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='课程具体的学习单元';

-- ======================================================================
-- 4. 技能点定义表 (Skills)
-- 对应 UI 中的 "HTML", "Grep", "Sed" 等细分知识点
-- ======================================================================
CREATE TABLE `skills` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `name` varchar(64) NOT NULL DEFAULT '' COMMENT '技能名 (唯一)',
  `category` varchar(32) NOT NULL DEFAULT 'General' COMMENT '分类 (如 Linux, Frontend)',
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_skill_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='原子技能/知识点库';

-- ======================================================================
-- 5. 课程单元与技能关联表 (Lesson Skills)
-- 解决 "Shell" 课程包含 "26 Skills" 的统计逻辑
-- ======================================================================
CREATE TABLE `lesson_skills` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `lesson_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '关联 course_lessons.id',
  `skill_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '关联 skills.id',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_lesson_skill` (`lesson_id`, `skill_id`),
  KEY `idx_skill_lesson` (`skill_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='单元技能映射表';

-- ======================================================================
-- 6. 用户课程进度表 (User Course Enrollment)
-- 对应 UI "Continue Learning" 和 "Welcome back" 面板
-- ======================================================================
CREATE TABLE `user_course_enrollments` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `user_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '关联 sys_users.id',
  `course_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '关联 courses.id',
  
  -- 进度概览
  `status` tinyint unsigned NOT NULL DEFAULT '1' COMMENT '1-进行中, 2-已完成',
  `progress_percent` tinyint unsigned NOT NULL DEFAULT '0' COMMENT '0-100',
  `completed_lessons_count` int unsigned NOT NULL DEFAULT '0' COMMENT '已完成课时数',
  
  -- "继续学习" 核心字段
  `last_lesson_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '最后学习的课时ID',
  `last_active_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '最后活跃时间(排序用)',
  
  `joined_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `completed_at` datetime DEFAULT NULL,
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_course` (`user_id`, `course_id`),
  KEY `idx_user_recent` (`user_id`, `last_active_at`) COMMENT '用于Dashboard获取"最近学习"'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户课程注册与宏观进度';

-- ======================================================================
-- 7. 用户课时明细记录 (User Lesson Records)
-- 记录每一个 Lab/Challenge 的完成情况
-- ======================================================================
CREATE TABLE `user_lesson_records` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `user_id` bigint unsigned NOT NULL DEFAULT '0',
  `course_id` bigint unsigned NOT NULL DEFAULT '0', -- 冗余，方便按课程统计
  `lesson_id` bigint unsigned NOT NULL DEFAULT '0',
  
  `status` tinyint unsigned NOT NULL DEFAULT '0' COMMENT '0-未开始, 1-进行中, 2-已完成',
  `score` int unsigned NOT NULL DEFAULT '0' COMMENT '得分/评级',
  `started_at` datetime DEFAULT NULL,
  `finished_at` datetime DEFAULT NULL,
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_lesson` (`user_id`, `lesson_id`),
  KEY `idx_course_stats` (`user_id`, `course_id`, `status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户具体课时完成记录';

-- ======================================================================
-- 8. 用户每日活动表 (User Activity Logs)
-- 对应 UI 中的 热力图 (Heatmap) 和 连续学习天数计算
-- ======================================================================
CREATE TABLE `user_activity_logs` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `user_id` bigint unsigned NOT NULL DEFAULT '0',
  `activity_date` date NOT NULL COMMENT '日期 (YYYY-MM-DD)',
  `contribution_count` int unsigned NOT NULL DEFAULT '1' COMMENT '当日活跃次数(完成实验数)',
  
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_date` (`user_id`, `activity_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户活跃热力图数据';

-- ======================================================================
-- 9. 课时学习资料表 (Lesson Materials)
-- 用于知识型课时(type=4)配置学习资料，支持 Markdown、PDF、思维导图
-- ======================================================================
CREATE TABLE `lesson_materials` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `lesson_id` bigint unsigned NOT NULL DEFAULT '0' COMMENT '关联 course_lessons.id',
  
  `title` varchar(128) NOT NULL DEFAULT '' COMMENT '资料标题',
  `type` tinyint unsigned NOT NULL DEFAULT '1' COMMENT '类型: 1-Markdown, 2-PDF, 3-思维导图',
  `file_url` varchar(512) NOT NULL DEFAULT '' COMMENT '文件URL路径',
  `file_name` varchar(256) NOT NULL DEFAULT '' COMMENT '原始文件名',
  `file_size` int unsigned NOT NULL DEFAULT '0' COMMENT '文件大小(字节)',
  `sort_order` int unsigned NOT NULL DEFAULT '0' COMMENT '排序权重',
  
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  PRIMARY KEY (`id`),
  KEY `idx_lesson_materials` (`lesson_id`, `sort_order`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='课时学习资料表';

```

#### A. 个人中心 Dashboard ("Welcome back, Seso20!")

1. **获取“继续学习”卡片 (Continue Learning)**:
   - **SQL逻辑**: 查询 `user_course_enrollments` 表。
   - **条件**: `WHERE user_id = ? ORDER BY last_active_at DESC LIMIT 1`。
   - **返回数据**:
     - 课程名称 (`courses.title`)。
     - 当前进度条 (`progress_percent`)。
     - 下一课时名称：通过 `last_lesson_id` 去 `course_lessons` 表查。如果 `last_lesson_id` 对应的状态是“已完成”，则需要查找该 Lesson 的 `sort_order` 下一个 Lesson。
2. **获取热力图数据 (Heatmap)**:
   - **SQL逻辑**: 查询 `user_activity_logs`。
   - **条件**: `WHERE user_id = ? AND activity_date >= (当前日期 - 1年)`。
   - **前端渲染**: 直接将 `{ date: '2025-01-01', count: 5 }` 数组传给前端组件。
3. **技能树列表 (All Skill Trees)**:
   - **逻辑**: 这里需要混合数据。
   - 第一步：查出所有 `courses`。
   - 第二步：查出当前用户在这些课程中的状态（LEFT JOIN `user_course_enrollments`）。
   - **UI状态判断**:
     - 如果 `enrollment` 记录存在 -> 显示 "已加入" 或进度。
     - 如果不存在 -> 显示 "加入" 按钮。

#### B. 课程详情页 ("Shell" 详情)

1. **构建大纲树 (The Syllabus)**:

   - 这是一个典型的**树形结构封装**问题。

   - **查询 1**: `SELECT * FROM course_stages WHERE course_id = ? ORDER BY sort_order`.

   - **查询 2**: `SELECT * FROM course_lessons WHERE course_id = ? ORDER BY sort_order`.

   - **后端组装**:

     TypeScript

     ```
     // 伪代码
     const stages = await db.getStages(courseId);
     const lessons = await db.getLessons(courseId);
     
     // 你的任务是将扁平的 lessons 塞入对应的 stage 中
     const tree = stages.map(stage => ({
         ...stage,
         modules: lessons.filter(l => l.stage_id === stage.id)
     }));
     ```

2. **计算锁定/解锁状态**:

   - UI 中可能要求“必须先完成 Lesson 1 才能看 Lesson 2”。
   - **逻辑**: 遍历 `tree`。检查用户 `user_lesson_records`。如果 Lesson N-1 未完成（无记录或 status != 2），则 Lesson N 的 `is_locked = true`。

#### C. 完成实验/挑战的触发器逻辑

当用户在前端点击“完成实验”或通过后端 Check 接口时：

1. **事务开启**:
2. **更新明细**: `INSERT INTO user_lesson_records ... ON DUPLICATE KEY UPDATE status=2, finished_at=NOW()`.
3. **更新活跃度**: `INSERT INTO user_activity_logs ... ON DUPLICATE KEY UPDATE contribution_count = contribution_count + 1`.
4. **计算课程总进度**:
   - `done_count` = `SELECT COUNT(*) FROM user_lesson_records WHERE course_id = ? AND status = 2`.
   - `total_count` = `courses.total_lessons`.
   - `percent` = `(done_count / total_count) * 100`.
5. **更新注册表**: `UPDATE user_course_enrollments SET progress_percent = ?, completed_lessons_count = ?, last_lesson_id = ?`.
6. **事务提交**.
