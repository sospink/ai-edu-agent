### 创建简化的输入护栏

```bash
cat > /workspace/courses/guardrails/input_rails.py << 'EOF'
#!/usr/bin/env python3
"""
输入护栏演示
展示如何检查用户输入
"""

class InputRails:
    """输入护栏"""
    
    def __init__(self):
        # 违禁词列表
        self.blocked_words = [
            "暴力", "色情", "赌博",
            "毒品", "诈骗"
        ]
        
        # 敏感词列表
        self.sensitive_words = [
            "密码", "账号", "身份证"
        ]
    
    def check(self, user_input):
        """
        检查用户输入
        
        Args:
            user_input: 用户输入
        
        Returns:
            检查结果
        """
        result = {
            "allowed": True,
            "reason": None,
            "warnings": []
        }
        
        # 检查1: 违禁词
        for word in self.blocked_words:
            if word in user_input:
                result["allowed"] = False
                result["reason"] = f"包含违禁词: {word}"
                return result
        
        # 检查2: 敏感词
        for word in self.sensitive_words:
            if word in user_input:
                result["warnings"].append(f"包含敏感词: {word}")
        
        # 检查3: 长度限制
        if len(user_input) > 500:
            result["allowed"] = False
            result["reason"] = "输入过长"
            return result
        
        # 检查4: 空输入
        if not user_input.strip():
            result["allowed"] = False
            result["reason"] = "输入为空"
            return result
        
        return result

# 测试
if __name__ == "__main__":
    print("=" * 60)
    print("输入护栏测试")
    print("=" * 60)
    print()
    
    rails = InputRails()
    
    test_inputs = [
        "你好，我想咨询一下",
        "我的密码是123456",
        "如何进行赌博",
        "",
        "a" * 600
    ]
    
    for i, user_input in enumerate(test_inputs, 1):
        print(f"测试 {i}: {user_input[:50]}...")
        result = rails.check(user_input)
        
        if result["allowed"]:
            print("✓ 通过")
            if result["warnings"]:
                print(f"  警告: {result['warnings']}")
        else:
            print(f"✗ 拒绝: {result['reason']}")
        print()
EOF
```

## 步骤 4: 实现 Output Rails（输出护栏）

### 什么是 Output Rails？

Output Rails 检查 AI 输出是否安全合规。

### 创建输出护栏

```bash
cat > /workspace/courses/guardrails/output_rails.py << 'EOF'
#!/usr/bin/env python3
"""
输出护栏演示
展示如何检查 AI 输出
"""

class OutputRails:
    """输出护栏"""
    
    def __init__(self):
        # 危险短语
        self.dangerous_phrases = [
            "保证", "一定", "绝对",
            "100%", "必然"
        ]
        
        # 需要免责声明的关键词
        self.disclaimer_keywords = [
            "医疗", "法律", "投资",
            "健康", "诊断"
        ]
    
    def check(self, ai_output):
        """
        检查 AI 输出
        
        Args:
            ai_output: AI 生成的输出
        
        Returns:
            检查结果
        """
        result = {
            "allowed": True,
            "modified_output": ai_output,
            "warnings": []
        }
        
        # 检查1: 危险短语
        for phrase in self.dangerous_phrases:
            if phrase in ai_output:
                result["allowed"] = False
                result["reason"] = f"包含危险短语: {phrase}"
                return result
        
        # 检查2: 添加免责声明
        for keyword in self.disclaimer_keywords:
            if keyword in ai_output:
                if "免责声明" not in ai_output:
                    result["modified_output"] += "\n\n⚠️ 免责声明：以上内容仅供参考，不构成专业建议。"
                    result["warnings"].append("已添加免责声明")
                break
        
        # 检查3: 长度检查
        if len(ai_output) > 2000:
            result["warnings"].append("输出较长，建议分段")
        
        return result

# 测试
if __name__ == "__main__":
    print("=" * 60)
    print("输出护栏测试")
    print("=" * 60)
    print()
    
    rails = OutputRails()
    
    test_outputs = [
        "这是一个正常的回复",
        "我保证这个方法一定有效",
        "根据您的症状，可能是感冒",
        "这是一个很长的回复..." + "x" * 2000
    ]
    
    for i, output in enumerate(test_outputs, 1):
        print(f"测试 {i}: {output[:50]}...")
        result = rails.check(output)
        
        if result["allowed"]:
            print("✓ 通过")
            if result["warnings"]:
                print(f"  警告: {result['warnings']}")
            if result["modified_output"] != output:
                print("  输出已修改")
        else:
            print(f"✗ 拒绝: {result['reason']}")
        print()
EOF
```

## 步骤 5: 实现 Dialog Rails（对话护栏）

### 什么是 Dialog Rails？

Dialog Rails 控制对话流程和状态。

### 创建对话护栏

```bash
cat > /workspace/courses/guardrails/dialog_rails.py << 'EOF'
#!/usr/bin/env python3
"""
对话护栏演示
展示如何控制对话流程
"""

class DialogRails:
    """对话护栏"""
    
    def __init__(self):
        self.max_turns = 20  # 最大对话轮数
        self.max_requests = 5  # 最大请求次数
        
        self.user_state = {
            "verified": False,
            "request_count": 0,
            "turn_count": 0
        }
    
    def check(self, conversation_history):
        """
        检查对话状态
        
        Args:
            conversation_history: 对话历史
        
        Returns:
            检查结果
        """
        result = {
            "allowed": True,
            "action": None,
            "reason": None
        }
        
        # 更新轮数
        self.user_state["turn_count"] = len(conversation_history)
        
        # 检查1: 对话轮数限制
        if self.user_state["turn_count"] >= self.max_turns:
            result["allowed"] = False
            result["action"] = "end_conversation"
            result["reason"] = "对话时间过长"
            return result
        
        # 检查2: 请求次数限制
        if self.user_state["request_count"] >= self.max_requests:
            result["allowed"] = False
            result["action"] = "limit_reached"
            result["reason"] = "请求次数已达上限"
            return result
        
        # 检查3: 身份验证
        if not self.user_state["verified"]:
            result["action"] = "request_verification"
            result["reason"] = "需要身份验证"
        
        return result
    
    def update_state(self, action):
        """更新用户状态"""
        if action == "verify":
            self.user_state["verified"] = True
        elif action == "request":
            self.user_state["request_count"] += 1

# 测试
if __name__ == "__main__":
    print("=" * 60)
    print("对话护栏测试")
    print("=" * 60)
    print()
    
    rails = DialogRails()
    conversation = []
    
    # 模拟对话
    scenarios = [
        ("用户请求1", "request"),
        ("用户请求2", "request"),
        ("用户请求3", "request"),
        ("用户请求4", "request"),
        ("用户请求5", "request"),
        ("用户请求6", "request"),  # 应该被限制
    ]
    
    for i, (message, action) in enumerate(scenarios, 1):
        print(f"轮次 {i}: {message}")
        conversation.append(message)
        
        result = rails.check(conversation)
        
        if result["allowed"]:
            print(f"✓ 允许 (action: {result['action']})")
            rails.update_state(action)
        else:
            print(f"✗ 拒绝: {result['reason']}")
        
        print(f"  状态: 请求次数={rails.user_state['request_count']}, "
              f"对话轮数={rails.user_state['turn_count']}")
        print()
EOF
```

### 创建完整系统

```bash
cat > /workspace/courses/guardrails/complete_guardrails.py << 'EOF'
#!/usr/bin/env python3
"""
完整的护栏系统
整合所有护栏类型
"""

from input_rails import InputRails
from output_rails import OutputRails
from dialog_rails import DialogRails

class CompleteGuardrails:
    """完整的护栏系统"""
    
    def __init__(self):
        self.input_rails = InputRails()
        self.output_rails = OutputRails()
        self.dialog_rails = DialogRails()
        self.conversation_history = []
    
    def process(self, user_input):
        """
        处理用户请求（完整的护栏流程）
        
        Args:
            user_input: 用户输入
        
        Returns:
            最终回复
        """
        print("\n" + "=" * 60)
        print(f"处理请求: {user_input}")
        print("=" * 60)
        
        # 1. 输入护栏检查
        print("\n[1] Input Rails 检查...")
        input_result = self.input_rails.check(user_input)
        
        if not input_result["allowed"]:
            return f"❌ 输入被拒绝: {input_result['reason']}"
        
        if input_result["warnings"]:
            print(f"⚠️  警告: {input_result['warnings']}")
        
        # 2. 对话护栏检查
        print("\n[2] Dialog Rails 检查...")
        dialog_result = self.dialog_rails.check(self.conversation_history)
        
        if not dialog_result["allowed"]:
            return f"❌ 对话被限制: {dialog_result['reason']}"
        
        if dialog_result["action"]:
            print(f"ℹ️  动作: {dialog_result['action']}")
        
        # 3. 生成 AI 回复（模拟）
        print("\n[3] 生成 AI 回复...")
        ai_response = self._generate_response(user_input)
        
        # 4. 输出护栏检查
        print("\n[4] Output Rails 检查...")
        output_result = self.output_rails.check(ai_response)
        
        if not output_result["allowed"]:
            return f"❌ 输出被拒绝: {output_result['reason']}"
        
        if output_result["warnings"]:
            print(f"⚠️  警告: {output_result['warnings']}")
        
        # 5. 更新对话历史
        self.conversation_history.append({
            "user": user_input,
            "assistant": output_result["modified_output"]
        })
        
        return output_result["modified_output"]
    
    def _generate_response(self, user_input):
        """生成 AI 回复（模拟）"""
        if "咨询" in user_input:
            return "我可以为您提供一些建议"
        elif "医疗" in user_input:
            return "关于医疗问题，建议咨询专业医生"
        else:
            return "我理解您的问题，让我来帮助您"

# 测试
if __name__ == "__main__":
    print("=" * 60)
    print("完整护栏系统测试")
    print("=" * 60)
    
    system = CompleteGuardrails()
    
    test_inputs = [
        "你好，我想咨询一下",
        "关于医疗问题",
        "我的密码是123456",
    ]
    
    for user_input in test_inputs:
        response = system.process(user_input)
        print(f"\n[最终回复]")
        print(response)
        print()
EOF
```
